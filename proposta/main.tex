% Os apêndices são material de consulta pessoal e NÃO integram o repositório

% entrega, sem erro de compilação.
%
% Trechos do texto que só fazem sentido quando os apêndices existem devem ser

% O Manual de Normalização da UFPR (2024) exige espaçamento simples no interior de
% tabelas e na lista de referências — só o corpo do texto usa 1,5. A classe não
% aplica essa distinção, então ela é feita aqui.
\AtBeginEnvironment{table}{\setstretch{1.0}}

% Alíneas conforme item 3.4.3 do Manual de Normalização da UFPR (2024):
% ordenadas por letras minúsculas seguidas de parêntese, texto iniciando em
% minúscula e terminando em ponto-e-vírgula (exceto a última, com ponto final).
\newlist{alineas}{enumerate}{1}
\setlist[alineas]{label=\alph*), leftmargin=1.5cm, itemsep=0.3\baselineskip,
                  topsep=0.5\baselineskip, parsep=0pt}

% Subalíneas (item 3.4.4): travessão, recuo maior.
\newlist{subalineas}{itemize}{1}
\setlist[subalineas]{label=\textendash, leftmargin=2.5cm, itemsep=0.2\baselineskip,
                     topsep=0.3\baselineskip, parsep=0pt}
